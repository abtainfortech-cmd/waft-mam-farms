    20→interface Farm { id: string; name: string; flocks: any[] }
    21→interface Vaccination { id: string; vaccineName: string; scheduledDate: string; administeredDate: string | null; status: string; method: string; flock: { name: string; birdType: string; birdCount: number }; farm: { name: string } }
    22→interface Treatment { id: string; date: string; diagnosis: string; medication: string; dosage: string; duration: string; cost: number; status: string; flock: { name: string }; farm: { name: string } }
    23→interface HealthCheck { id: string; date: string; overallStatus: string; symptoms: string | null; bodyCondition: string | null; temperature: number | null; recommendations: string | null; flock: { name: string; birdType: string }; farm: { name: string } }
    24→
    25→export function VetDashboard() {
    26→  const { selectedFarmId, setFarm } = useAppStore()
    27→  const [farms, setFarms] = useState<Farm[]>([])
    28→  const [vaccinations, setVaccinations] = useState<Vaccination[]>([])
    29→  const [treatments, setTreatments] = useState<Treatment[]>([])
    30→  const [healthChecks, setHealthChecks] = useState<HealthCheck[]>([])
    31→  const [loading, setLoading] = useState(true)
    32→
    33→  const fetchData = useCallback(async () => {
    34→    setLoading(true)
    35→    try {
    36→      const [farmsRes, vacRes, treatRes, healthRes] = await Promise.all([
    37→        fetch('/api/farms'),
    38→        fetch(`/api/vaccinations${selectedFarmId ? `?farmId=${selectedFarmId}` : ''}`),
    39→        fetch(`/api/treatments${selectedFarmId ? `?farmId=${selectedFarmId}` : ''}`),
    40→        fetch(`/api/health${selectedFarmId ? `?farmId=${selectedFarmId}` : ''}`),
    41→      ])
    42→      if (farmsRes.ok) setFarms(await farmsRes.json())
    43→      if (vacRes.ok) setVaccinations(await vacRes.json())
    44→      if (treatRes.ok) setTreatments(await treatRes.json())
    45→      if (healthRes.ok) setHealthChecks(await healthRes.json())
    46→    } catch (e) { console.error(e) }
    47→    finally { setLoading(false) }
    48→  }, [selectedFarmId])
    49→
    50→  useEffect(() => { fetchData() }, [fetchData])
    51→
    52→  const now = new Date()
    53→  const overdue = vaccinations.filter(v => new Date(v.scheduledDate) <= now && v.status !== 'Completed')
    54→  const upcoming = vaccinations.filter(v => {
    55→    const d = new Date(v.scheduledDate)
    56→    const twoWeeks = new Date(now.getTime() + 14 * 86400000)
    57→    return d > now && d <= twoWeeks && v.status !== 'Completed'
    58→  })
    59→  const completed = vaccinations.filter(v => v.status === 'Completed')
    60→  const activeTreatments = treatments.filter(t => t.status === 'Ongoing')
    61→
    62→  const today = new Date().toISOString().split('T')[0]
    63→
    64→  const submitHealthCheck = async (e: React.FormEvent<HTMLFormElement>) => {
    65→    e.preventDefault()
    66→    const f = e.currentTarget
    67→    const data = {
    68→      flockId: (f.elements.namedItem('flockId') as HTMLSelectElement).value,
    69→      farmId: (f.elements.namedItem('farmId') as HTMLSelectElement).value,
    70→      date: today,
    71→      overallStatus: (f.elements.namedItem('overallStatus') as HTMLSelectElement).value,
    72→      symptoms: (f.elements.namedItem('symptoms') as HTMLInputElement).value || null,
    73→      temperature: parseFloat((f.elements.namedItem('temperature') as HTMLInputElement).value) || null,
    74→      bodyCondition: (f.elements.namedItem('bodyCondition') as HTMLSelectElement).value || null,
    75→      recommendations: (f.elements.namedItem('recommendations') as HTMLTextAreaElement).value || null,
    76→      vetOfficer: 'Vet Officer',
    77→    }
    78→    const res = await fetch('/api/health', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data) })
    79→    if (res.ok) { toast.success('Health check recorded!'); f.reset(); fetchData() }
    80→    else toast.error('Failed to save')
    81→  }
    82→
    83→  const submitVaccination = async (e: React.FormEvent<HTMLFormElement>) => {
    84→    e.preventDefault()
    85→    const f = e.currentTarget
    86→    const data = {
    87→      flockId: (f.elements.namedItem('flockId') as HTMLSelectElement).value,
    88→      farmId: (f.elements.namedItem('farmId') as HTMLSelectElement).value,
    89→      vaccineName: (f.elements.namedItem('vaccineName') as HTMLSelectElement).value,
    90→      scheduledDate: (f.elements.namedItem('scheduledDate') as HTMLInputElement).value || today,
    91→      status: 'Scheduled',
    92→      method: (f.elements.namedItem('method') as HTMLSelectElement).value || null,
    93→      dosage: (f.elements.namedItem('dosage') as HTMLInputElement).value || null,
    94→      cost: parseFloat((f.elements.namedItem('cost') as HTMLInputElement).value) || null,
    95→      batchNo: (f.elements.namedItem('batchNo') as HTMLInputElement).value || null,
    96→    }
    97→    const res = await fetch('/api/vaccinations', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data) })
    98→    if (res.ok) { toast.success('Vaccination scheduled!'); f.reset(); fetchData() }
    99→    else toast.error('Failed to schedule')
   100→  }
   101→
   102→  const submitTreatment = async (e: React.FormEvent<HTMLFormElement>) => {
   103→    e.preventDefault()
   104→    const f = e.currentTarget
   105→    const data = {
   106→      flockId: (f.elements.namedItem('flockId') as HTMLSelectElement).value,
   107→      farmId: (f.elements.namedItem('farmId') as HTMLSelectElement).value,

   109→      diagnosis: (f.elements.namedItem('diagnosis') as HTMLInputElement).value,
   110→      medication: (f.elements.namedItem('medication') as HTMLInputElement).value,

   115→      vetOfficer: 'Vet Officer',
   116→    }

   119→    else toast.error('Failed to save')

   129→    else toast.error('Failed to update')
   130→  }
   131→
   132→  if (loading && farms.length === 0) return <div className="grid grid-cols-2 gap-3">{Array.from({ length: 4 }).map((_, i) => <Skeleton key={i} className="h-28 rounded-xl" />)}</div>
   133→

   138→      case 'Completed': return 'default'
   139→      default: return 'secondary'
   140→    }
   141→  }
   142→

   178→            <div className="flex items-center gap-2 mb-1">
   179→              <Shield className="h-4 w-4 text-red-600" />
   180→              <span className="text-xs text-gray-500">Overdue</span>
   181→            </div>
   182→            <p className="text-2xl font-bold text-red-600">{overdue.length}</p>
   183→            <p className="text-xs text-gray-500">vaccinations behind schedule</p>
   184→          </CardContent>
   185→        </Card>
   186→        <Card>
   187→          <CardContent className="p-4">
   188→            <div className="flex items-center gap-2 mb-1">
   189→              <Clock className="h-4 w-4 text-amber-600" />
   190→              <span className="text-xs text-gray-500">Upcoming (14 days)</span>
   191→            </div>
   192→            <p className="text-2xl font-bold text-amber-600">{upcoming.length}</p>
   193→            <p className="text-xs text-gray-500">vaccinations scheduled</p>
   194→          </CardContent>
   195→        </Card>
   196→        <Card>
   197→          <CardContent className="p-4">
   198→            <div className="flex items-center gap-2 mb-1">
   199→              <Activity className="h-4 w-4 text-purple-600" />
   200→              <span className="text-xs text-gray-500">Active Treatments</span>
   201→            </div>
   202→            <p className="text-2xl font-bold text-purple-600">{activeTreatments.length}</p>
   203→            <p className="text-xs text-gray-500">ongoing treatments</p>
   204→          </CardContent>
   205→        </Card>
   206→        <Card>
   207→          <CardContent className="p-4">
   208→            <div className="flex items-center gap-2 mb-1">
   209→              <Heart className="h-4 w-4 text-green-600" />
   210→              <span className="text-xs text-gray-500">Health Checks</span>
   211→            </div>
   212→            <p className="text-2xl font-bold">{healthChecks.length}</p>
   213→            <p className="text-xs text-gray-500">total recorded</p>
   214→          </CardContent>
   215→        </Card>
   216→      </div>
   217→
   218→      {/* Farm Filter */}
   219→      <div className="flex items-center gap-2">
   220→        <Label className="text-xs text-gray-500">Farm:</Label>
   221→        <Select value={selectedFarmId || 'all'} onValueChange={(v) => setFarm(v === 'all' ? null : v)}>
   222→          <SelectTrigger className="w-48 h-8 text-xs"><SelectValue placeholder="All Farms" /></SelectTrigger>
   223→          <SelectContent>
   224→            <SelectItem value="all">All Farms</SelectItem>
   225→            {farms.map(f => <SelectItem key={f.id} value={f.id}>{f.name}</SelectItem>)}
   226→          </SelectContent>
   227→        </Select>
   228→      </div>
   229→
   230→      <Tabs defaultValue="schedule" className="w-full">
   231→        <TabsList className="grid grid-cols-4 w-full">
   232→          <TabsTrigger value="schedule"><Syringe className="h-3 w-3 mr-1" />Vaccinations</TabsTrigger>
   233→          <TabsTrigger value="health"><Stethoscope className="h-3 w-3 mr-1" />Health</TabsTrigger>
   234→          <TabsTrigger value="treatments"><Activity className="h-3 w-3 mr-1" />Treatments</TabsTrigger>
   235→          <TabsTrigger value="history"><CheckCircle2 className="h-3 w-3 mr-1" />History</TabsTrigger>
   236→        </TabsList>
   237→
   238→        <TabsContent value="schedule">
   239→          <Card>
   240→            <CardHeader className="pb-2">
   241→              <CardTitle className="text-base">Schedule Vaccination</CardTitle>
   242→              <CardDescription>Add a new vaccination to the schedule</CardDescription>
   243→            </CardHeader>
   244→            <CardContent>
   245→              <form onSubmit={submitVaccination} className="grid grid-cols-2 md:grid-cols-3 gap-3">
   246→                <div>
   247→                  <Label className="text-xs">Farm *</Label>
   248→                  <Select name="farmId" required>
   249→                    <SelectTrigger className="h-9 text-sm"><SelectValue placeholder="Select farm" /></SelectTrigger>
   250→                    <SelectContent>
   251→                      {farms.map(f => <SelectItem key={f.id} value={f.id}>{f.name}</SelectItem>)}
   252→                    </SelectContent>
   253→                  </Select>
   254→                </div>

   260→                      {farms.flatMap(f => f.flocks.map(fl => <SelectItem key={fl.id} value={fl.id}>{fl.name} ({f.name})</SelectItem>))}

   262→                  </Select>

   264→                <div>
   265→                  <Label className="text-xs">Vaccine *</Label>
   266→                  <Select name="vaccineName" required>
   267→                    <SelectTrigger className="h-9 text-sm"><SelectValue placeholder="Select vaccine" /></SelectTrigger>
   268→                    <SelectContent>
   269→                      {['Newcastle Disease', 'Gumboro', 'Fowl Pox', 'Fowl Typhoid', 'Infectious Bronchitis', 'Coryza', 'Newcastle Booster', 'Marek\'s Disease', 'Coccidiosis'].map(v => (
   270→                        <SelectItem key={v} value={v}>{v}</SelectItem>
   271→                      ))}
   272→                    </SelectContent>
   273→                  </Select>
   274→                </div>
   275→                <div>
   276→                  <Label className="text-xs">Scheduled Date</Label>
   277→                  <Input name="scheduledDate" type="date" defaultValue={today} className="h-9 text-sm" />
   278→                </div>
   279→                <div>
   280→                  <Label className="text-xs">Method</Label>
   281→                  <Select name="method">
