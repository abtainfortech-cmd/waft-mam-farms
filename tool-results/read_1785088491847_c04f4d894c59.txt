     1→'use client'
     2→
     3→import { useEffect, useState, useCallback } from 'react'
     4→import { Card, CardContent, CardDescription, CardHeader, CardTitle } from '@/components/ui/card'
     5→import { Badge } from '@/components/ui/badge'
     6→import { Button } from '@/components/ui/button'
     7→import { Input } from '@/components/ui/input'
     8→import { Label } from '@/components/ui/label'
     9→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select'
    10→import { Textarea } from '@/components/ui/textarea'
    11→import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs'
    12→import { ScrollArea } from '@/components/ui/scroll-area'
    13→import { Separator } from '@/components/ui/separator'
    14→import { Skeleton } from '@/components/ui/skeleton'
    15→import { useAppStore } from '@/store/app'
    16→import { toast } from 'sonner'
    17→import { LiveSyncIndicator } from '@/components/farm/LiveSyncIndicator'
    18→import {
    19→  ShoppingBasket, Plus, Bird, DollarSign, Users, AlertCircle, CheckCircle, CreditCard, Package, Pencil
    20→} from 'lucide-react'
    21→import { AmendmentRequestDialog } from '@/components/farm/AmendmentRequestDialog'
    22→
    23→interface Farm { id: string; name: string }
    24→interface Customer { id: string; name: string; phone: string; location: string; customerType: string }
    25→interface EggSale { id: string; date: string; crateCount: number; eggsPerCrate: number; pricePerCrate: number; totalAmount: number; paymentStatus: string; amountPaid: number; customer: { name: string } | null }
    26→interface BirdSale { id: string; date: string; quantity: number; pricePerBird: number; totalAmount: number; paymentStatus: string; amountPaid: number; customer: { name: string } | null }
    27→
    28→function formatGHS(n: number) { return `GHS ${n.toLocaleString('en-GH', { minimumFractionDigits: 2, maximumFractionDigits: 2 })}` }
    29→
    30→export function SalesDashboard() {
    31→  const { selectedFarmId, setFarm, currentUser } = useAppStore()
    32→  const [farms, setFarms] = useState<Farm[]>([])
    33→  const [customers, setCustomers] = useState<Customer[]>([])
    34→  const [eggSales, setEggSales] = useState<EggSale[]>([])
    35→  const [birdSales, setBirdSales] = useState<BirdSale[]>([])
    36→  const [loading, setLoading] = useState(true)
    37→  const [amendmentOpen, setAmendmentOpen] = useState(false)
    38→  const [amendmentTarget, setAmendmentTarget] = useState<{ recordType: string; recordId: string; fields: any[] }>({ recordType: '', recordId: '', fields: [] })
    39→
    40→  const today = new Date().toISOString().split('T')[0]
    41→
    42→  // Form Select state (Radix Select doesn't use native form elements)
    43→  const [eggSaleCustomerId, setEggSaleCustomerId] = useState('')
    44→  const [eggSaleFarmId, setEggSaleFarmId] = useState(selectedFarmId || '')
    45→  const [eggSalePayment, setEggSalePayment] = useState('Paid')
    46→  const [eggAmountPaid, setEggAmountPaid] = useState('')
    47→  const [birdSaleCustomerId, setBirdSaleCustomerId] = useState('')
    48→  const [birdSaleFarmId, setBirdSaleFarmId] = useState(selectedFarmId || '')
    49→  const [birdSalePayment, setBirdSalePayment] = useState('Paid')
    50→  const [birdAmountPaid, setBirdAmountPaid] = useState('')
    51→  const [customerType, setCustomerType] = useState('Retail')
    52→
    53→  // Controlled numerical input state for egg sale form
    54→  const [eggCrateCount, setEggCrateCount] = useState('')
    55→  const [eggPricePerCrate, setEggPricePerCrate] = useState('')
    56→  const [eggSaleDate, setEggSaleDate] = useState(today)
    57→
    58→  // Controlled numerical input state for bird sale form
    59→  const [birdQuantity, setBirdQuantity] = useState('')
    60→  const [birdPricePerBird, setBirdPricePerBird] = useState('')
    61→  const [birdWeightPerBird, setBirdWeightPerBird] = useState('')
    62→  const [birdSaleDate, setBirdSaleDate] = useState(today)
    63→
    64→  // Controlled input state for customer form
    65→  const [custName, setCustName] = useState('')
    66→  const [custPhone, setCustPhone] = useState('')
    67→  const [custLocation, setCustLocation] = useState('')
    68→
    69→  useEffect(() => {
    70→    if (selectedFarmId) {
    71→      setEggSaleFarmId(selectedFarmId)
    72→      setBirdSaleFarmId(selectedFarmId)
    73→    }
    74→  }, [selectedFarmId])
    75→
    76→  const fetchData = useCallback(async () => {
    77→    setLoading(true)
    78→    try {
    79→      const [farmsRes, custRes, eggsRes, birdsRes] = await Promise.all([
    80→        fetch('/api/farms'), fetch('/api/customers'),
    81→        fetch(`/api/sales/eggs${selectedFarmId ? `?farmId=${selectedFarmId}` : ''}`),
    82→        fetch(`/api/sales/birds${selectedFarmId ? `?farmId=${selectedFarmId}` : ''}`),
    83→      ])
    84→      if (farmsRes.ok) setFarms(await farmsRes.json())
    85→      if (custRes.ok) setCustomers(await custRes.json())
    86→      if (eggsRes.ok) setEggSales(await eggsRes.json())
    87→      if (birdsRes.ok) setBirdSales(await birdsRes.json())
    88→    } catch (e) { console.error(e) }
    89→    finally { setLoading(false) }
    90→  }, [selectedFarmId])
    91→
    92→  useEffect(() => { fetchData() }, [fetchData])
    93→
    94→  // Auto-refresh: Sales sees new records from farm hand, accountant etc.
    95→  const handleAutoData = useCallback((results: any[]) => {
    96→    if (results && results.length >= 4) {
    97→      setFarms(results[0] || [])
    98→      setCustomers(results[1] || [])
    99→      setEggSales(results[2] || [])
   100→      setBirdSales(results[3] || [])
   101→    }
   102→  }, [])
   103→
   104→  const monthAgo = new Date(Date.now() - 30 * 86400000).toISOString().split('T')[0]
   105→  const monthEggSales = eggSales.filter(e => new Date(e.date).toISOString().split('T')[0] >= monthAgo)
   106→  const monthBirdSales = birdSales.filter(e => new Date(e.date).toISOString().split('T')[0] >= monthAgo)
   107→  const totalEggRev = monthEggSales.reduce((s, e) => s + e.totalAmount, 0)
   108→  const totalBirdRev = monthBirdSales.reduce((s, e) => s + e.totalAmount, 0)
   109→  const unpaidEgg = monthEggSales.filter(e => e.paymentStatus !== 'Paid').reduce((s, e) => s + (e.totalAmount - e.amountPaid), 0)
   110→  const unpaidBird = monthBirdSales.filter(e => e.paymentStatus !== 'Paid').reduce((s, e) => s + (e.totalAmount - e.amountPaid), 0)
   111→
   112→  const submitEggSale = async (e: React.FormEvent<HTMLFormElement>) => {
   113→    e.preventDefault()
   114→    try {
   115→      const crates = Number(eggCrateCount)
   116→      const price = Number(eggPricePerCrate)
   117→      if (!crates || crates <= 0) { toast.error('Please enter a valid crate count'); return }
   118→      if (!price || price <= 0) { toast.error('Please enter a valid price per crate'); return }
   119→      if (!eggSaleFarmId) { toast.error('Please select a farm'); return }
   120→      const data = {
   121→        customerId: eggSaleCustomerId || null,
   122→        date: eggSaleDate || today,
   123→        crateCount: crates,
   124→        eggsPerCrate: 30,
   125→        pricePerCrate: price,
   126→        totalAmount: crates * price,
   127→        paymentStatus: eggSalePayment,
   128→        amountPaid: eggSalePayment === 'Paid' ? crates * price : eggSalePayment === 'Partial' ? (Number(eggAmountPaid) || 0) : 0,
   129→        farmId: eggSaleFarmId,
   130→        recordedBy: currentUser?.name || 'Sales',
   131→      }
   132→      const res = await fetch('/api/sales/eggs', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data) })
   133→      if (res.ok) {
   134→        toast.success('Egg sale recorded!')
   135→        setEggCrateCount(''); setEggPricePerCrate(''); setEggAmountPaid('')
   136→        setEggSaleCustomerId(''); setEggSalePayment('Paid')
   137→        fetchData()
   138→      } else {
   139→        const err = await res.json().catch(() => ({}))
   140→        toast.error(err.error || 'Failed to record sale')
   141→      }
   142→    } catch (err) { console.error(err); toast.error('Failed to record sale') }
   143→  }
   144→
   145→  const submitBirdSale = async (e: React.FormEvent<HTMLFormElement>) => {
   146→    e.preventDefault()
   147→    try {
   148→      const qty = Number(birdQuantity)
   149→      const price = Number(birdPricePerBird)
   150→      if (!qty || qty <= 0) { toast.error('Please enter a valid quantity'); return }
   151→      if (!price || price <= 0) { toast.error('Please enter a valid price per bird'); return }
   152→      if (!birdSaleFarmId) { toast.error('Please select a farm'); return }
   153→      const data = {
   154→        customerId: birdSaleCustomerId || null,
   155→        date: birdSaleDate || today,
   156→        quantity: qty,
   157→        weightPerBirdKg: birdWeightPerBird ? Number(birdWeightPerBird) : null,
   158→        pricePerBird: price,
   159→        totalAmount: qty * price,
   160→        paymentStatus: birdSalePayment,
   161→        amountPaid: birdSalePayment === 'Paid' ? qty * price : birdSalePayment === 'Partial' ? (Number(birdAmountPaid) || 0) : 0,
   162→        farmId: birdSaleFarmId,
   163→        recordedBy: currentUser?.name || 'Sales',
   164→      }
   165→      const res = await fetch('/api/sales/birds', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data) })
   166→      if (res.ok) {
   167→        toast.success('Bird sale recorded!')
   168→        setBirdQuantity(''); setBirdPricePerBird(''); setBirdWeightPerBird(''); setBirdAmountPaid('')
   169→        setBirdSaleCustomerId(''); setBirdSalePayment('Paid')
   170→        fetchData()
   171→      } else {
   172→        const err = await res.json().catch(() => ({}))
   173→        toast.error(err.error || 'Failed to record sale')
   174→      }
   175→    } catch (err) { console.error(err); toast.error('Failed to record sale') }
   176→  }
   177→
   178→  const openAmendmentDialog = (recordType: string, recordId: string, fields: any[]) => {
   179→    setAmendmentTarget({ recordType, recordId, fields })
   180→    setAmendmentOpen(true)
   181→  }
   182→
   183→  const addCustomer = async (e: React.FormEvent<HTMLFormElement>) => {
   184→    e.preventDefault()
   185→    try {
   186→      const data = {
   187→        name: custName.trim(),
   188→        phone: custPhone.trim() || null,
   189→        location: custLocation.trim() || null,
   190→        customerType: customerType,
   191→      }
   192→      if (!data.name) { toast.error('Please enter customer name'); return }
   193→      const res = await fetch('/api/customers', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data) })
   194→      if (res.ok) {
   195→        toast.success('Customer added!')
   196→        setCustName(''); setCustPhone(''); setCustLocation('')
   197→        fetchData()
   198→      } else {
   199→        const err = await res.json().catch(() => ({}))
   200→        toast.error(err.error || 'Failed to add customer')
   201→      }
   202→    } catch (err) { console.error(err); toast.error('Failed to add customer') }
   203→  }
   204→
   205→  return (
   206→    <div className="space-y-4 md:space-y-6">
   207→      {/* Stats */}
   208→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   209→        <Card>
   210→          <CardContent className="p-4">
   211→            <div className="flex items-center gap-2 mb-1">
   212→              <ShoppingBasket className="h-4 w-4 text-green-600" />
   213→              <span className="text-xs text-gray-500">Egg Sales (Month)</span>
   214→            </div>
   215→            <p className="text-xl font-bold text-green-700">{formatGHS(totalEggRev)}</p>
   216→            <p className="text-xs text-gray-500">{monthEggSales.length} transactions</p>
   217→          </CardContent>
   218→        </Card>
   219→        <Card>
   220→          <CardContent className="p-4">
   221→            <div className="flex items-center gap-2 mb-1">
   222→              <Bird className="h-4 w-4 text-blue-600" />
   223→              <span className="text-xs text-gray-500">Bird Sales (Month)</span>
   224→            </div>
   225→            <p className="text-xl font-bold text-blue-700">{formatGHS(totalBirdRev)}</p>
   226→            <p className="text-xs text-gray-500">{monthBirdSales.length} transactions</p>
   227→          </CardContent>
   228→        </Card>
   229→        <Card>
   230→          <CardContent className="p-4">
   231→            <div className="flex items-center gap-2 mb-1">
   232→              <DollarSign className="h-4 w-4 text-amber-600" />
   233→              <span className="text-xs text-gray-500">Total Revenue</span>
   234→            </div>
   235→            <p className="text-xl font-bold">{formatGHS(totalEggRev + totalBirdRev)}</p>
   236→          </CardContent>
   237→        </Card>
   238→        <Card>
   239→          <CardContent className="p-4">
   240→            <div className="flex items-center gap-2 mb-1">
   241→              <CreditCard className="h-4 w-4 text-red-600" />
   242→              <span className="text-xs text-gray-500">Outstanding</span>
   243→            </div>
   244→            <p className="text-xl font-bold text-red-600">{formatGHS(unpaidEgg + unpaidBird)}</p>
   245→          </CardContent>
   246→        </Card>
   247→      </div>
   248→
   249→      {/* Farm Filter */}
   250→      <div className="flex items-center gap-2">
   251→        <Label className="text-xs text-gray-500">Farm:</Label>
   252→        <Select value={selectedFarmId || 'all'} onValueChange={(v) => setFarm(v === 'all' ? null : v)}>
   253→          <SelectTrigger className="w-48 h-8 text-xs"><SelectValue placeholder="All Farms" /></SelectTrigger>
   254→          <SelectContent>
   255→            <SelectItem value="all">All Farms</SelectItem>
   256→            {farms.map(f => <SelectItem key={f.id} value={f.id}>{f.name}</SelectItem>)}
   257→          </SelectContent>
   258→        </Select>
   259→      </div>
   260→
   261→      <LiveSyncIndicator
   262→        endpoints={['/api/farms', '/api/customers', `/api/sales/eggs${selectedFarmId ? `?farmId=${selectedFarmId}` : ''}`, `/api/sales/birds${selectedFarmId ? `?farmId=${selectedFarmId}` : ''}`]}
   263→        interval={20000}
   264→        onData={handleAutoData}
   265→        compact
   266→      />
   267→
   268→      <Tabs defaultValue="egg-sales" className="w-full">
   269→        <TabsList className="grid grid-cols-4 w-full">
   270→          <TabsTrigger value="egg-sales"><ShoppingBasket className="h-3 w-3 mr-1" />Egg Sales</TabsTrigger>
   271→          <TabsTrigger value="bird-sales"><Bird className="h-3 w-3 mr-1" />Bird Sales</TabsTrigger>
   272→          <TabsTrigger value="customers"><Users className="h-3 w-3 mr-1" />Customers</TabsTrigger>
   273→          <TabsTrigger value="unpaid"><AlertCircle className="h-3 w-3 mr-1" />Unpaid</TabsTrigger>
   274→        </TabsList>
   275→
   276→        <TabsContent value="egg-sales">
   277→          <Card>
   278→            <CardHeader className="pb-2">
   279→              <CardTitle className="text-base">Record Egg Sale</CardTitle>
   280→            </CardHeader>
   281→            <CardContent>
   282→              <form onSubmit={submitEggSale} className="grid grid-cols-2 md:grid-cols-3 gap-3">
   283→                <div>
   284→                  <Label className="text-xs">Customer</Label>
   285→                  <Select value={eggSaleCustomerId} onValueChange={setEggSaleCustomerId}>
   286→                    <SelectTrigger className="h-9 text-sm"><SelectValue placeholder="Walk-in" /></SelectTrigger>
   287→                    <SelectContent>
   288→                      {customers.map(c => <SelectItem key={c.id} value={c.id}>{c.name}</SelectItem>)}
   289→                    </SelectContent>
   290→                  </Select>
   291→                </div>
   292→                <div>
   293→                  <Label className="text-xs">Farm *</Label>
   294→                  <Select value={eggSaleFarmId} onValueChange={setEggSaleFarmId}>
   295→                    <SelectTrigger className="h-9 text-sm"><SelectValue placeholder="Select farm" /></SelectTrigger>
   296→                    <SelectContent>
   297→                      {farms.map(f => <SelectItem key={f.id} value={f.id}>{f.name}</SelectItem>)}
   298→                    </SelectContent>
   299→                  </Select>
   300→                </div>
   301→                <div>
   302→                  <Label className="text-xs">Date</Label>
   303→                  <Input type="date" value={eggSaleDate} onChange={e => setEggSaleDate(e.target.value)} className="h-9 text-sm" />
   304→                </div>
   305→                <div>
   306→                  <Label className="text-xs">Crates *</Label>
   307→                  <Input type="number" min="1" required placeholder="e.g. 10" value={eggCrateCount} onChange={e => setEggCrateCount(e.target.value)} className="h-9 text-sm" />
   308→                </div>
   309→                <div>
   310→                  <Label className="text-xs">Price per Crate (GHS) *</Label>
   311→                  <Input type="number" min="0" step="0.5" required placeholder="e.g. 35" value={eggPricePerCrate} onChange={e => setEggPricePerCrate(e.target.value)} className="h-9 text-sm" />
   312→                </div>
   313→                <div>
   314→                  <Label className="text-xs">Payment</Label>
   315→                  <Select value={eggSalePayment} onValueChange={(v) => { setEggSalePayment(v); if (v !== 'Partial') setEggAmountPaid('') }}>
   316→                    <SelectTrigger className="h-9 text-sm"><SelectValue /></SelectTrigger>
   317→                    <SelectContent>
   318→                      <SelectItem value="Paid">Paid</SelectItem>
   319→                      <SelectItem value="Partial">Partial</SelectItem>
   320→                      <SelectItem value="Unpaid">Credit / Unpaid</SelectItem>
   321→                    </SelectContent>
   322→                  </Select>
   323→                </div>
   324→                {eggSalePayment === 'Partial' && (
   325→                  <div className="col-span-2 md:col-span-1">
   326→                    <Label className="text-xs">Amount Paid (GHS)</Label>
   327→                    <Input type="number" min="0" step="0.5" value={eggAmountPaid} onChange={e => setEggAmountPaid(e.target.value)} placeholder="e.g. 500" className="h-9 text-sm" />
   328→                  </div>
   329→                )}
   330→                <div className="flex items-end col-span-2 md:col-span-3">
   331→                  <Button type="submit" className="w-full md:w-auto h-9"><Plus className="h-3 w-3 mr-1" />Record Sale</Button>
   332→                </div>
   333→              </form>
   334→            </CardContent>
   335→          </Card>
   336→
   337→          {/* Recent egg sales */}
   338→          <Card className="mt-4">
   339→            <CardHeader className="pb-2">
   340→              <CardTitle className="text-base">Recent Egg Sales</CardTitle>
   341→            </CardHeader>
   342→            <CardContent>
   343→              <ScrollArea className="max-h-64">
   344→                <div className="space-y-2">
   345→                  {eggSales.slice(0, 20).map(s => (
   346→                    <div key={s.id} className="flex items-center gap-3 p-2 bg-gray-50 rounded-lg text-sm">
   347→                      <ShoppingBasket className="h-4 w-4 text-green-600 shrink-0" />
   348→                      <div className="flex-1">
   349→                        <p className="font-medium">{s.customer?.name || 'Walk-in'}</p>
   350→                        <p className="text-xs text-gray-500">{new Date(s.date).toLocaleDateString('en-GB')} · {s.crateCount} crates x {formatGHS(s.pricePerCrate)}</p>
   351→                      </div>
   352→                      <div className="text-right">
   353→                        <p className="font-medium">{formatGHS(s.totalAmount)}</p>
   354→                        <Badge variant={s.paymentStatus === 'Paid' ? 'default' : 'destructive'} className="text-[10px]">
   355→                          {s.paymentStatus}
   356→                        </Badge>
   357→                        {s.paymentStatus !== 'Paid' && s.totalAmount > 0 && (
   358→                          <p className="text-[10px] text-red-600 mt-0.5">
   359→                            Balance: {formatGHS(s.totalAmount - s.amountPaid)}
   360→                          </p>
   361→                        )}
   362→                      </div>
   363→                      <Button size="sm" variant="ghost" className="h-7 w-7 p-0 shrink-0 text-gray-400 hover:text-amber-600" onClick={() => openAmendmentDialog('EggSale', s.id, [
   364→                        { key: 'crateCount', label: 'Crates', type: 'number' as const, value: s.crateCount },
   365→                        { key: 'pricePerCrate', label: 'Price/Crate', type: 'number' as const, value: s.pricePerCrate },
   366→                        { key: 'paymentStatus', label: 'Payment Status', type: 'text' as const, value: s.paymentStatus },
   367→                        { key: 'amountPaid', label: 'Amount Paid', type: 'number' as const, value: s.amountPaid },
   368→                      ])}>
   369→                        <Pencil className="h-3 w-3" />
   370→                      </Button>
   371→                    </div>
   372→                  ))}
   373→                </div>
   374→              </ScrollArea>
   375→            </CardContent>
   376→          </Card>
   377→        </TabsContent>
   378→
   379→        <TabsContent value="bird-sales">
   380→          <Card>
   381→            <CardHeader className="pb-2">
   382→              <CardTitle className="text-base">Record Bird Sale</CardTitle>
   383→            </CardHeader>
   384→            <CardContent>
   385→              <form onSubmit={submitBirdSale} className="grid grid-cols-2 md:grid-cols-3 gap-3">
   386→                <div>
   387→                  <Label className="text-xs">Customer</Label>
   388→                  <Select value={birdSaleCustomerId} onValueChange={setBirdSaleCustomerId}>
   389→                    <SelectTrigger className="h-9 text-sm"><SelectValue placeholder="Walk-in" /></SelectTrigger>
   390→                    <SelectContent>
   391→                      {customers.map(c => <SelectItem key={c.id} value={c.id}>{c.name}</SelectItem>)}
   392→                    </SelectContent>
   393→                  </Select>
   394→                </div>
   395→                <div>
   396→                  <Label className="text-xs">Farm *</Label>
   397→                  <Select value={birdSaleFarmId} onValueChange={setBirdSaleFarmId}>
   398→                    <SelectTrigger className="h-9 text-sm"><SelectValue placeholder="Select farm" /></SelectTrigger>
   399→                    <SelectContent>
   400→                      {farms.map(f => <SelectItem key={f.id} value={f.id}>{f.name}</SelectItem>)}
   401→                    </SelectContent>
   402→                  </Select>
   403→                </div>
   404→                <div>
   405→                  <Label className="text-xs">Date</Label>
   406→                  <Input type="date" value={birdSaleDate} onChange={e => setBirdSaleDate(e.target.value)} className="h-9 text-sm" />
   407→                </div>
   408→                <div>
   409→                  <Label className="text-xs">Quantity *</Label>
   410→                  <Input type="number" min="1" required placeholder="e.g. 50" value={birdQuantity} onChange={e => setBirdQuantity(e.target.value)} className="h-9 text-sm" />
   411→                </div>
   412→                <div>
   413→                  <Label className="text-xs">Price per Bird (GHS) *</Label>
   414→                  <Input type="number" min="0" step="0.5" required placeholder="e.g. 30" value={birdPricePerBird} onChange={e => setBirdPricePerBird(e.target.value)} className="h-9 text-sm" />
   415→                </div>
   416→                <div>
