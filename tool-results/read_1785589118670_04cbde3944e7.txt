    24→interface Farm { id: string; name: string; flocks: any[] }
    25→interface FlockWithAge { id: string; name: string; birdType: string; breed: string | null; birdCount: number; farmId: string; farm: { name: string }; currentAgeWeeks: number }
    26→interface Vaccination { id: string; vaccineName: string; scheduledDate: string; administeredDate: string | null; status: string; method: string; flock: { name: string; birdType: string; birdCount: number }; farm: { name: string } }

    29→
    30→export function VetDashboard() {
    31→  const { selectedFarmId, setFarm, currentUser } = useAppStore()
    32→  const [farms, setFarms] = useState<Farm[]>([])
    33→  const [allFlocks, setAllFlocks] = useState<FlockWithAge[]>([])
    34→  const [vaccinations, setVaccinations] = useState<Vaccination[]>([])
    35→  const [treatments, setTreatments] = useState<Treatment[]>([])
    36→  const [healthChecks, setHealthChecks] = useState<HealthCheck[]>([])
    37→  const [loading, setLoading] = useState(true)
    38→  const [amendmentOpen, setAmendmentOpen] = useState(false)
    39→  const [amendmentTarget, setAmendmentTarget] = useState<{ recordType: string; recordId: string; fields: any[] }>({ recordType: '', recordId: '', fields: [] })
    40→
    41→  // Form Select state (Radix Select doesn't use native form elements)
    42→  const [healthFarmId, setHealthFarmId] = useState(selectedFarmId || '')
    43→  const [healthFlockId, setHealthFlockId] = useState('')
    44→  const [healthStatus, setHealthStatus] = useState('')
    45→  const [healthBodyCondition, setHealthBodyCondition] = useState('')
    46→  const [vacFarmId, setVacFarmId] = useState(selectedFarmId || '')
    47→  const [vacFlockId, setVacFlockId] = useState('')
    48→  const [vacVaccineName, setVacVaccineName] = useState('')
    49→  const [vacMethod, setVacMethod] = useState('')
    50→  const [treatFarmId, setTreatFarmId] = useState(selectedFarmId || '')
    51→  const [treatFlockId, setTreatFlockId] = useState('')
    52→
    53→  // Controlled input state for health check form
    54→  const [healthSymptoms, setHealthSymptoms] = useState('')
    55→  const [healthTemperature, setHealthTemperature] = useState('')
    56→  const [healthRecommendations, setHealthRecommendations] = useState('')
    57→
    58→  // Controlled input state for vaccination form
    59→  const [vacScheduledDate, setVacScheduledDate] = useState(new Date().toISOString().split('T')[0])
    60→  const [vacDosage, setVacDosage] = useState('')
    61→  const [vacCost, setVacCost] = useState('')
    62→  const [vacBatchNo, setVacBatchNo] = useState('')
    63→
    64→  // Controlled input state for treatment form
    65→  const [treatDiagnosis, setTreatDiagnosis] = useState('')
    66→  const [treatMedication, setTreatMedication] = useState('')
    67→  const [treatDosage, setTreatDosage] = useState('')
    68→  const [treatDuration, setTreatDuration] = useState('')
    69→  const [treatCost, setTreatCost] = useState('')
    70→
    71→  useEffect(() => {
    72→    if (selectedFarmId) {
    73→      setHealthFarmId(selectedFarmId)
    74→      setVacFarmId(selectedFarmId)
    75→      setTreatFarmId(selectedFarmId)
    76→    }
    77→  }, [selectedFarmId])
    78→
    79→  const fetchData = useCallback(async () => {
    80→    setLoading(true)
    81→    try {
    82→      const [farmsRes, flocksRes, vacRes, treatRes, healthRes] = await Promise.all([
    83→        fetch('/api/farms'),
    84→        fetch(`/api/flocks`),
    85→        fetch(`/api/vaccinations${selectedFarmId ? `?farmId=${selectedFarmId}` : ''}`),
    86→        fetch(`/api/treatments${selectedFarmId ? `?farmId=${selectedFarmId}` : ''}`),
    87→        fetch(`/api/health${selectedFarmId ? `?farmId=${selectedFarmId}` : ''}`),
    88→      ])
    89→      if (farmsRes.ok) setFarms(await farmsRes.json())
    90→      if (flocksRes.ok) setAllFlocks(await flocksRes.json())
    91→      if (vacRes.ok) setVaccinations(await vacRes.json())
    92→      if (treatRes.ok) setTreatments(await treatRes.json())
    93→      if (healthRes.ok) setHealthChecks(await healthRes.json())
    94→    } catch (e) { console.error(e) }
    95→    finally { setLoading(false) }
    96→  }, [selectedFarmId])
    97→
    98→  useEffect(() => { fetchData() }, [fetchData])
    99→
   100→  // Auto-refresh: Vet sees mortality and flock data from farm hand in real-time
   101→  const handleAutoData = useCallback((results: any[]) => {
   102→    if (results && results.length >= 5) {
   103→      setFarms(results[0] || [])
   104→      setAllFlocks(results[1] || [])
   105→      setVaccinations(results[2] || [])
   106→      setTreatments(results[3] || [])
   107→      setHealthChecks(results[4] || [])
   108→    }
   109→  }, [])
   110→
   111→  const now = new Date()
   112→  const overdue = vaccinations.filter(v => new Date(v.scheduledDate) <= now && v.status !== 'Completed')
   113→  const upcoming = vaccinations.filter(v => {
   114→    const d = new Date(v.scheduledDate)
   115→    const twoWeeks = new Date(now.getTime() + 14 * 86400000)
   116→    return d > now && d <= twoWeeks && v.status !== 'Completed'
   117→  })
   118→  const completed = vaccinations.filter(v => v.status === 'Completed')
   119→  const activeTreatments = treatments.filter(t => t.status === 'Ongoing')
   120→
   121→  const today = new Date().toISOString().split('T')[0]
   122→
   123→  const submitHealthCheck = async (e: React.FormEvent<HTMLFormElement>) => {
   124→    e.preventDefault()
   125→    try {
   126→      const data = {
   127→        flockId: healthFlockId,
   128→        farmId: healthFarmId,
   129→        date: today,
   130→        overallStatus: healthStatus,
   131→        symptoms: healthSymptoms.trim() || null,
   132→        temperature: healthTemperature ? Number(healthTemperature) : null,
   133→        bodyCondition: healthBodyCondition || null,
   134→        recommendations: healthRecommendations.trim() || null,
   135→        vetOfficer: currentUser?.name || 'Vet Officer',
   136→      }
   137→      if (!data.farmId || !data.flockId || !data.overallStatus) { toast.error('Please select farm, flock and status'); return }
   138→      const res = await fetch('/api/health', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data) })
   139→      if (res.ok) {
   140→        toast.success('Health check recorded!')
   141→        setHealthSymptoms(''); setHealthTemperature(''); setHealthRecommendations('')
   142→        setHealthStatus(''); setHealthBodyCondition(''); setHealthFlockId('')
   143→        fetchData()
   144→      } else {
   145→        const err = await res.json().catch(() => ({}))
   146→        toast.error(err.error || 'Failed to save')
   147→      }
   148→    } catch (err) { console.error(err); toast.error('Failed to save health check') }
   149→  }
   150→
   151→  const submitVaccination = async (e: React.FormEvent<HTMLFormElement>) => {
   152→    e.preventDefault()
   153→    try {
   154→      const data = {
   155→        flockId: vacFlockId,
   156→        farmId: vacFarmId,
   157→        vaccineName: vacVaccineName,
   158→        scheduledDate: vacScheduledDate || today,
   159→        status: 'Scheduled',
   160→        method: vacMethod || null,
   161→        dosage: vacDosage.trim() || null,
   162→        cost: vacCost ? Number(vacCost) : null,
   163→        batchNo: vacBatchNo.trim() || null,
   164→      }
   165→      if (!data.farmId || !data.flockId || !data.vaccineName) { toast.error('Please select farm, flock and vaccine'); return }
   166→      const res = await fetch('/api/vaccinations', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data) })
   167→      if (res.ok) {
   168→        toast.success('Vaccination scheduled!')
   169→        setVacVaccineName(''); setVacMethod(''); setVacDosage(''); setVacCost(''); setVacBatchNo('')
   170→        setVacFlockId('')
   171→        fetchData()
   172→      } else {
   173→        const err = await res.json().catch(() => ({}))
   174→        toast.error(err.error || 'Failed to schedule')
   175→      }
   176→    } catch (err) { console.error(err); toast.error('Failed to schedule vaccination') }
   177→  }
   178→
   179→  const submitTreatment = async (e: React.FormEvent<HTMLFormElement>) => {
   180→    e.preventDefault()
   181→    try {
   182→      const data = {
   183→        flockId: treatFlockId,
   184→        farmId: treatFarmId,
   185→        date: today,
   186→        diagnosis: treatDiagnosis.trim(),
   187→        medication: treatMedication.trim(),
   188→        dosage: treatDosage.trim() || null,
   189→        duration: treatDuration.trim() || null,
   190→        cost: treatCost ? Number(treatCost) : null,
   191→        status: 'Ongoing',
   192→        vetOfficer: currentUser?.name || 'Vet Officer',
   193→      }
   194→      if (!data.farmId || !data.flockId || !data.diagnosis || !data.medication) { toast.error('Please select farm, flock, diagnosis and medication'); return }
   195→      const res = await fetch('/api/treatments', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data) })
   196→      if (res.ok) {
   197→        toast.success('Treatment recorded!')
   198→        setTreatDiagnosis(''); setTreatMedication(''); setTreatDosage(''); setTreatDuration(''); setTreatCost('')
   199→        setTreatFlockId('')
   200→        fetchData()
   201→      } else {
   202→        const err = await res.json().catch(() => ({}))
   203→        toast.error(err.error || 'Failed to save')
   204→      }
   205→    } catch (err) { console.error(err); toast.error('Failed to save treatment') }
   206→  }
   207→
   208→  const markVaccinationDone = async (id: string) => {
   209→    const res = await fetch('/api/vaccinations', {
   210→      method: 'PUT',
   211→      headers: { 'Content-Type': 'application/json' },
   212→      body: JSON.stringify({ id, status: 'Completed', administeredDate: today }),
   213→    })
   214→    if (res.ok) { toast.success('Vaccination marked as completed'); fetchData() }
   215→    else toast.error('Failed to update')
   216→  }
   217→
   218→  const openAmendmentDialog = (recordType: string, recordId: string, fields: any[]) => {
   219→    setAmendmentTarget({ recordType, recordId, fields })
   220→    setAmendmentOpen(true)
   221→  }
   222→
   223→  if (loading && farms.length === 0) return <div className="grid grid-cols-2 gap-3">{Array.from({ length: 4 }).map((_, i) => <Skeleton key={i} className="h-28 rounded-xl" />)}</div>
   224→
   225→  const statusColor = (status: string) => {
   226→    switch (status) {
   227→      case 'Overdue': return 'destructive'
   228→      case 'Scheduled': return 'secondary'
   229→      case 'Completed': return 'default'
   230→      default: return 'secondary'
   231→    }
   232→  }
   233→
   234→  return (
   235→    <div className="space-y-4 md:space-y-6">
   236→      {/* Critical Alerts */}
   237→      {overdue.length > 0 && (
   238→        <Card className="border-red-200 bg-red-50">
   239→          <CardContent className="p-3">
   240→            <div className="flex items-center gap-2 mb-2">
   241→              <AlertTriangle className="h-5 w-5 text-red-600" />
   242→              <p className="text-sm font-bold text-red-800">{overdue.length} Overdue Vaccination{overdue.length > 1 ? 's' : ''}</p>
   243→            </div>
   244→            <ScrollArea className="max-h-32">
   245→              <div className="space-y-1">
   246→                {overdue.slice(0, 5).map(v => (
   247→                  <div key={v.id} className="flex items-center gap-2 text-xs">
   248→                    <Badge variant="destructive" className="text-[10px]">Overdue</Badge>
   249→                    <span>{v.vaccineName}</span>
   250→                    <span className="text-gray-500">·</span>
   251→                    <span>{v.flock?.name}</span>
   252→                    <span className="text-gray-500">·</span>
   253→                    <span>{v.farm?.name}</span>
   254→                    <Button size="sm" variant="outline" className="h-6 ml-auto text-[10px] px-2" onClick={() => markVaccinationDone(v.id)}>
   255→                      Mark Done
   256→                    </Button>
   257→                  </div>
   258→                ))}
   259→              </div>
   260→            </ScrollArea>
   261→          </CardContent>
   262→        </Card>
   263→      )}
   264→
   265→      {/* KPI Cards */}
   266→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   267→        <Card>
   268→          <CardContent className="p-4">
   269→            <div className="flex items-center gap-2 mb-1">
   270→              <Shield className="h-4 w-4 text-red-600" />
   271→              <span className="text-xs text-gray-500">Overdue</span>
   272→            </div>
   273→            <p className="text-2xl font-bold text-red-600">{overdue.length}</p>
   274→            <p className="text-xs text-gray-500">vaccinations behind schedule</p>
   275→          </CardContent>
   276→        </Card>
   277→        <Card>
   278→          <CardContent className="p-4">
   279→            <div className="flex items-center gap-2 mb-1">
   280→              <Clock className="h-4 w-4 text-amber-600" />
   281→              <span className="text-xs text-gray-500">Upcoming (14 days)</span>
   282→            </div>
   283→            <p className="text-2xl font-bold text-amber-600">{upcoming.length}</p>
   284→            <p className="text-xs text-gray-500">vaccinations scheduled</p>
   285→          </CardContent>
   286→        </Card>
   287→        <Card>
   288→          <CardContent className="p-4">
   289→            <div className="flex items-center gap-2 mb-1">
   290→              <Activity className="h-4 w-4 text-purple-600" />
   291→              <span className="text-xs text-gray-500">Active Treatments</span>
   292→            </div>
   293→            <p className="text-2xl font-bold text-purple-600">{activeTreatments.length}</p>
   294→            <p className="text-xs text-gray-500">ongoing treatments</p>
   295→          </CardContent>
   296→        </Card>
   297→        <Card>
   298→          <CardContent className="p-4">
   299→            <div className="flex items-center gap-2 mb-1">
   300→              <Heart className="h-4 w-4 text-green-600" />
   301→              <span className="text-xs text-gray-500">Health Checks</span>
   302→            </div>
   303→            <p className="text-2xl font-bold">{healthChecks.length}</p>
   304→            <p className="text-xs text-gray-500">total recorded</p>
   305→          </CardContent>
   306→        </Card>
   307→      </div>
   308→
   309→      {/* Farm Filter */}
   310→      <div className="flex items-center gap-2">
   311→        <Label className="text-xs text-gray-500">Farm:</Label>
   312→        <Select value={selectedFarmId || 'all'} onValueChange={(v) => setFarm(v === 'all' ? null : v)}>
   313→          <SelectTrigger className="w-48 h-8 text-xs"><SelectValue placeholder="All Farms" /></SelectTrigger>
   314→          <SelectContent>
   315→            <SelectItem value="all">All Farms</SelectItem>
   316→            {farms.map(f => <SelectItem key={f.id} value={f.id}>{f.name}</SelectItem>)}
   317→          </SelectContent>
   318→        </Select>
   319→      </div>
   320→
   321→      <LiveSyncIndicator
   322→        endpoints={['/api/farms', '/api/flocks', `/api/vaccinations${selectedFarmId ? `?farmId=${selectedFarmId}` : ''}`, `/api/treatments${selectedFarmId ? `?farmId=${selectedFarmId}` : ''}`, `/api/health${selectedFarmId ? `?farmId=${selectedFarmId}` : ''}`]}
   323→        interval={20000}
   324→        onData={handleAutoData}
   325→        compact
   326→      />
   327→
   328→      <Tabs defaultValue="schedule" className="w-full">
   329→        <TabsList className="grid grid-cols-5 w-full">
   330→          <TabsTrigger value="schedule"><Syringe className="h-3 w-3 mr-1" />Vaccinations</TabsTrigger>
   331→          <TabsTrigger value="std-schedule"><BookOpen className="h-3 w-3 mr-1" />Standard Schedule</TabsTrigger>
   332→          <TabsTrigger value="health"><Stethoscope className="h-3 w-3 mr-1" />Health</TabsTrigger>
   333→          <TabsTrigger value="treatments"><Activity className="h-3 w-3 mr-1" />Treatments</TabsTrigger>
   334→          <TabsTrigger value="history"><CheckCircle2 className="h-3 w-3 mr-1" />History</TabsTrigger>
   335→        </TabsList>
   336→
   337→        <TabsContent value="schedule">
   338→          <Card>
   339→            <CardHeader className="pb-2">
   340→              <CardTitle className="text-base">Schedule Vaccination</CardTitle>
   341→              <CardDescription>Add a new vaccination to the schedule</CardDescription>
   342→            </CardHeader>
   343→            <CardContent>
   344→              <form onSubmit={submitVaccination} className="grid grid-cols-2 md:grid-cols-3 gap-3">
   345→                <div>
   346→                  <Label className="text-xs">Farm *</Label>
   347→                  <Select value={vacFarmId} onValueChange={setVacFarmId}>
   348→                    <SelectTrigger className="h-9 text-sm"><SelectValue placeholder="Select farm" /></SelectTrigger>
   349→                    <SelectContent>
   350→                      {farms.map(f => <SelectItem key={f.id} value={f.id}>{f.name}</SelectItem>)}
   351→                    </SelectContent>
   352→                  </Select>
   353→                </div>
   354→                <div>
   355→                  <Label className="text-xs">Flock *</Label>

   359→                      {farms.flatMap(f => f.flocks.map(fl => <SelectItem key={fl.id} value={fl.id}>{fl.name} ({f.name})</SelectItem>))}
   360→                    </SelectContent>
   361→                  </Select>
   362→                </div>
   363→                <div>
   364→                  <Label className="text-xs">Vaccine *</Label>
   365→                  <Select value={vacVaccineName} onValueChange={setVacVaccineName}>
   366→                    <SelectTrigger className="h-9 text-sm"><SelectValue placeholder="Select vaccine" /></SelectTrigger>
   367→                    <SelectContent>
   368→                      {['Newcastle Disease', 'Gumboro', 'Fowl Pox', 'Fowl Typhoid', 'Infectious Bronchitis', 'Coryza', 'Newcastle Booster', 'Marek\'s Disease', 'Coccidiosis'].map(v => (
   369→                        <SelectItem key={v} value={v}>{v}</SelectItem>
   370→                      ))}
   371→                    </SelectContent>
   372→                  </Select>
   373→                </div>
   374→                <div>
   375→                  <Label className="text-xs">Scheduled Date</Label>
   376→                  <Input type="date" value={vacScheduledDate} onChange={e => setVacScheduledDate(e.target.value)} className="h-9 text-sm" />
   377→                </div>
   378→                <div>
   379→                  <Label className="text-xs">Method</Label>
   380→                  <Select value={vacMethod} onValueChange={setVacMethod}>
   381→                    <SelectTrigger className="h-9 text-sm"><SelectValue placeholder="Method" /></SelectTrigger>
   382→                    <SelectContent>
   383→                      <SelectItem value="Eye drop">Eye Drop</SelectItem>
   384→                      <SelectItem value="Drinking water">Drinking Water</SelectItem>
   385→                      <SelectItem value="Injection">Injection</SelectItem>
   386→                      <SelectItem value="Spray">Spray</SelectItem>
   387→                      <SelectItem value="Wing web puncture">Wing Web Puncture</SelectItem>
   388→                    </SelectContent>
   389→                  </Select>
   390→                </div>
   391→                <div>
   392→                  <Label className="text-xs">Batch No.</Label>
   393→                  <Input placeholder="VAC-XXXX" value={vacBatchNo} onChange={e => setVacBatchNo(e.target.value)} className="h-9 text-sm" />
   394→                </div>
   395→                <div>
   396→                  <Label className="text-xs">Cost (GHS)</Label>
   397→                  <Input type="number" min="0" step="0.5" placeholder="e.g. 150" value={vacCost} onChange={e => setVacCost(e.target.value)} className="h-9 text-sm" />
   398→                </div>
   399→                <div>
   400→                  <Label className="text-xs">Dosage</Label>
   401→                  <Input placeholder="e.g. 0.5ml per bird" value={vacDosage} onChange={e => setVacDosage(e.target.value)} className="h-9 text-sm" />
   402→                </div>
   403→                <div className="flex items-end col-span-2 md:col-span-3">
   404→                  <Button type="submit" className="w-full md:w-auto h-9"><Plus className="h-3 w-3 mr-1" />Schedule Vaccination</Button>
   405→                </div>
   406→              </form>
   407→            </CardContent>
   408→          </Card>
   409→
   410→          {/* Vaccination List */}
   411→          <div className="grid grid-cols-1 md:grid-cols-2 gap-4 mt-4">
   412→            <Card>
   413→              <CardHeader className="pb-2">
   414→                <CardTitle className="text-base text-red-600">Overdue ({overdue.length})</CardTitle>
   415→              </CardHeader>
   416→              <CardContent>
   417→                <ScrollArea className="max-h-56">
   418→                  <div className="space-y-2">

   427→                          Done
   428→                        </Button>
   429→                      </div>
   430→                    ))}
   431→                    {overdue.length === 0 && <p className="text-center text-gray-400 py-3 text-xs">All up to date!</p>}
   432→                  </div>
