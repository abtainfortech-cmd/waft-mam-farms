     1→'use client'
     2→

     4→import { Card, CardContent, CardDescription, CardHeader, CardTitle } from '@/components/ui/card'
     5→import { Badge } from '@/components/ui/badge'
     6→import { Button } from '@/components/ui/button'
     7→import { Input } from '@/components/ui/input'
     8→import { Label } from '@/components/ui/label'
     9→import { Select, SelectContent, SelectItem, SelectTrigger, SelectValue } from '@/components/ui/select'
    10→import { Textarea } from '@/components/ui/textarea'
    11→import { Tabs, TabsContent, TabsList, TabsTrigger } from '@/components/ui/tabs'
    12→import { ScrollArea } from '@/components/ui/scroll-area'
    13→import { Separator } from '@/components/ui/separator'
    14→import { Skeleton } from '@/components/ui/skeleton'
    15→import { useAppStore } from '@/store/app'
    16→import { toast } from 'sonner'
    17→import { LiveSyncIndicator } from '@/components/farm/LiveSyncIndicator'
    18→import {
    19→  ShoppingBasket, Plus, Bird, DollarSign, Users, AlertCircle, CheckCircle, CreditCard, Package, Pencil
    20→} from 'lucide-react'
    21→import { AmendmentRequestDialog } from '@/components/farm/AmendmentRequestDialog'
    22→
    23→interface Farm { id: string; name: string }
    24→interface Customer { id: string; name: string; phone: string; location: string; customerType: string }
    25→interface EggSale { id: string; date: string; crateCount: number; eggsPerCrate: number; pricePerCrate: number; totalAmount: number; paymentStatus: string; amountPaid: number; customer: { name: string } | null }
    26→interface BirdSale { id: string; date: string; quantity: number; pricePerBird: number; totalAmount: number; paymentStatus: string; amountPaid: number; customer: { name: string } | null }
    27→
    28→function formatGHS(n: number) { return `GHS ${n.toLocaleString('en-GH', { minimumFractionDigits: 2, maximumFractionDigits: 2 })}` }
    29→
    30→export function SalesDashboard() {
    31→  const { selectedFarmId, setFarm, currentUser } = useAppStore()
    32→  const [farms, setFarms] = useState<Farm[]>([])
    33→  const [customers, setCustomers] = useState<Customer[]>([])
    34→  const [eggSales, setEggSales] = useState<EggSale[]>([])
    35→  const [birdSales, setBirdSales] = useState<BirdSale[]>([])
    36→  const [loading, setLoading] = useState(true)
    37→  const [amendmentOpen, setAmendmentOpen] = useState(false)
    38→  const [amendmentTarget, setAmendmentTarget] = useState<{ recordType: string; recordId: string; fields: any[] }>({ recordType: '', recordId: '', fields: [] })
    39→
    40→  // Form Select state (Radix Select doesn't use native form elements)
    41→  const [eggSaleCustomerId, setEggSaleCustomerId] = useState('')
    42→  const [eggSaleFarmId, setEggSaleFarmId] = useState(selectedFarmId || '')
    43→  const [eggSalePayment, setEggSalePayment] = useState('Paid')
    44→  const [birdSaleCustomerId, setBirdSaleCustomerId] = useState('')
    45→  const [birdSaleFarmId, setBirdSaleFarmId] = useState(selectedFarmId || '')
    46→  const [birdSalePayment, setBirdSalePayment] = useState('Paid')
    47→  const [customerType, setCustomerType] = useState('Retail')
    48→
    49→  useEffect(() => {
    50→    if (selectedFarmId) {
    51→      setEggSaleFarmId(selectedFarmId)
    52→      setBirdSaleFarmId(selectedFarmId)
    53→    }
    54→  }, [selectedFarmId])
    55→
    56→  const fetchData = useCallback(async () => {
    57→    setLoading(true)
    58→    try {
    59→      const [farmsRes, custRes, eggsRes, birdsRes] = await Promise.all([
    60→        fetch('/api/farms'), fetch('/api/customers'),
    61→        fetch(`/api/sales/eggs${selectedFarmId ? `?farmId=${selectedFarmId}` : ''}`),
    62→        fetch(`/api/sales/birds${selectedFarmId ? `?farmId=${selectedFarmId}` : ''}`),
    63→      ])
    64→      if (farmsRes.ok) setFarms(await farmsRes.json())
    65→      if (custRes.ok) setCustomers(await custRes.json())
    66→      if (eggsRes.ok) setEggSales(await eggsRes.json())
    67→      if (birdsRes.ok) setBirdSales(await birdsRes.json())
    68→    } catch (e) { console.error(e) }
    69→    finally { setLoading(false) }
    70→  }, [selectedFarmId])
    71→
    72→  useEffect(() => { fetchData() }, [fetchData])
    73→
    74→  // Auto-refresh: Sales sees new records from farm hand, accountant etc.
    75→  const handleAutoData = useCallback((results: any[]) => {
    76→    if (results && results.length >= 4) {
    77→      setFarms(results[0] || [])
    78→      setCustomers(results[1] || [])
    79→      setEggSales(results[2] || [])
    80→      setBirdSales(results[3] || [])
    81→    }
    82→  }, [])
    83→
    84→  const today = new Date().toISOString().split('T')[0]
    85→  const monthAgo = new Date(Date.now() - 30 * 86400000).toISOString().split('T')[0]
    86→  const monthEggSales = eggSales.filter(e => new Date(e.date).toISOString().split('T')[0] >= monthAgo)
    87→  const monthBirdSales = birdSales.filter(e => new Date(e.date).toISOString().split('T')[0] >= monthAgo)
    88→  const totalEggRev = monthEggSales.reduce((s, e) => s + e.totalAmount, 0)
    89→  const totalBirdRev = monthBirdSales.reduce((s, e) => s + e.totalAmount, 0)
    90→  const unpaidEgg = monthEggSales.filter(e => e.paymentStatus !== 'Paid').reduce((s, e) => s + (e.totalAmount - e.amountPaid), 0)
    91→  const unpaidBird = monthBirdSales.filter(e => e.paymentStatus !== 'Paid').reduce((s, e) => s + (e.totalAmount - e.amountPaid), 0)
    92→
    93→  const submitEggSale = async (e: React.FormEvent<HTMLFormElement>) => {
    94→    e.preventDefault()
    95→    const f = e.currentTarget
    96→    try {
    97→      const crates = parseInt((f.elements.namedItem('crateCount') as HTMLInputElement).value) || 0
    98→      const price = parseFloat((f.elements.namedItem('pricePerCrate') as HTMLInputElement).value) || 0
    99→      const data = {
   100→        customerId: eggSaleCustomerId || null,
   101→        date: (f.elements.namedItem('date') as HTMLInputElement).value || today,
   102→        crateCount: crates,
   103→        eggsPerCrate: parseInt((f.elements.namedItem('eggsPerCrate') as HTMLInputElement).value) || 30,
   104→        pricePerCrate: price,
   105→        totalAmount: crates * price,
   106→        paymentStatus: eggSalePayment,
   107→        amountPaid: eggSalePayment === 'Paid' ? crates * price : 0,
   108→        farmId: eggSaleFarmId,
   109→        recordedBy: 'Sales',
   110→      }
   111→      if (!data.farmId) { toast.error('Please select a farm'); return }
   112→      const res = await fetch('/api/sales/eggs', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data) })
   113→      if (res.ok) { toast.success('Egg sale recorded!'); f.reset(); fetchData() }
   114→      else { const err = await res.json().catch(() => ({})); toast.error(err.error || 'Failed to record sale') }
   115→    } catch (err) { console.error(err); toast.error('Failed to record sale') }
   116→  }
   117→
   118→  const submitBirdSale = async (e: React.FormEvent<HTMLFormElement>) => {
   119→    e.preventDefault()
   120→    const f = e.currentTarget
   121→    try {
   122→      const qty = parseInt((f.elements.namedItem('quantity') as HTMLInputElement).value) || 0
   123→      const price = parseFloat((f.elements.namedItem('pricePerBird') as HTMLInputElement).value) || 0
   124→      const data = {
   125→        customerId: birdSaleCustomerId || null,
   126→        date: (f.elements.namedItem('date') as HTMLInputElement).value || today,
   127→        quantity: qty,
   128→        weightPerBirdKg: parseFloat((f.elements.namedItem('weightPerBirdKg') as HTMLInputElement).value) || null,
   129→        pricePerBird: price,
   130→        totalAmount: qty * price,
   131→        paymentStatus: birdSalePayment,
   132→        amountPaid: birdSalePayment === 'Paid' ? qty * price : 0,
   133→        farmId: birdSaleFarmId,
   134→        recordedBy: 'Sales',
   135→      }
   136→      if (!data.farmId) { toast.error('Please select a farm'); return }
   137→      const res = await fetch('/api/sales/birds', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data) })
   138→      if (res.ok) { toast.success('Bird sale recorded!'); f.reset(); fetchData() }
   139→      else { const err = await res.json().catch(() => ({})); toast.error(err.error || 'Failed to record sale') }
   140→    } catch (err) { console.error(err); toast.error('Failed to record sale') }
   141→  }
   142→
   143→  const openAmendmentDialog = (recordType: string, recordId: string, fields: any[]) => {

   145→    setAmendmentOpen(true)
   146→  }
   147→
   148→  const addCustomer = async (e: React.FormEvent<HTMLFormElement>) => {
   149→    e.preventDefault()
   150→    const f = e.currentTarget
   151→    try {
   152→      const data = {
   153→        name: (f.elements.namedItem('name') as HTMLInputElement).value,
   154→        phone: (f.elements.namedItem('phone') as HTMLInputElement).value || null,
   155→        location: (f.elements.namedItem('location') as HTMLInputElement).value || null,
   156→        customerType: customerType,
   157→      }
   158→      if (!data.name) { toast.error('Please enter customer name'); return }
   159→      const res = await fetch('/api/customers', { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify(data) })
   160→      if (res.ok) { toast.success('Customer added!'); f.reset(); fetchData() }
   161→      else { const err = await res.json().catch(() => ({})); toast.error(err.error || 'Failed to add customer') }
   162→    } catch (err) { console.error(err); toast.error('Failed to add customer') }
   163→  }
   164→
   165→  return (
   166→    <div className="space-y-4 md:space-y-6">
   167→      {/* Stats */}
   168→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   169→        <Card>
   170→          <CardContent className="p-4">
   171→            <div className="flex items-center gap-2 mb-1">
   172→              <ShoppingBasket className="h-4 w-4 text-green-600" />
   173→              <span className="text-xs text-gray-500">Egg Sales (Month)</span>
   174→            </div>
   175→            <p className="text-xl font-bold text-green-700">{formatGHS(totalEggRev)}</p>
   176→            <p className="text-xs text-gray-500">{monthEggSales.length} transactions</p>
   177→          </CardContent>
   178→        </Card>
   179→        <Card>
   180→          <CardContent className="p-4">
   181→            <div className="flex items-center gap-2 mb-1">
   182→              <Bird className="h-4 w-4 text-blue-600" />
   183→              <span className="text-xs text-gray-500">Bird Sales (Month)</span>
   184→            </div>
   185→            <p className="text-xl font-bold text-blue-700">{formatGHS(totalBirdRev)}</p>
   186→            <p className="text-xs text-gray-500">{monthBirdSales.length} transactions</p>
   187→          </CardContent>
   188→        </Card>
   189→        <Card>
   190→          <CardContent className="p-4">
   191→            <div className="flex items-center gap-2 mb-1">
   192→              <DollarSign className="h-4 w-4 text-amber-600" />
   193→              <span className="text-xs text-gray-500">Total Revenue</span>
   194→            </div>
   195→            <p className="text-xl font-bold">{formatGHS(totalEggRev + totalBirdRev)}</p>
   196→          </CardContent>
   197→        </Card>
   198→        <Card>
   199→          <CardContent className="p-4">
   200→            <div className="flex items-center gap-2 mb-1">
   201→              <CreditCard className="h-4 w-4 text-red-600" />
   202→              <span className="text-xs text-gray-500">Outstanding</span>
   203→            </div>
   204→            <p className="text-xl font-bold text-red-600">{formatGHS(unpaidEgg + unpaidBird)}</p>
   205→          </CardContent>
   206→        </Card>
   207→      </div>
   208→
   209→      {/* Farm Filter */}
   210→      <div className="flex items-center gap-2">
   211→        <Label className="text-xs text-gray-500">Farm:</Label>
   212→        <Select value={selectedFarmId || 'all'} onValueChange={(v) => setFarm(v === 'all' ? null : v)}>
   213→          <SelectTrigger className="w-48 h-8 text-xs"><SelectValue placeholder="All Farms" /></SelectTrigger>
   214→          <SelectContent>
   215→            <SelectItem value="all">All Farms</SelectItem>
   216→            {farms.map(f => <SelectItem key={f.id} value={f.id}>{f.name}</SelectItem>)}
   217→          </SelectContent>
   218→        </Select>
   219→      </div>
   220→
   221→      <LiveSyncIndicator
   222→        endpoints={['/api/farms', '/api/customers', `/api/sales/eggs${selectedFarmId ? `?farmId=${selectedFarmId}` : ''}`, `/api/sales/birds${selectedFarmId ? `?farmId=${selectedFarmId}` : ''}`]}
   223→        interval={20000}
   224→        onData={handleAutoData}
   225→        compact
   226→      />
   227→
   228→      <Tabs defaultValue="egg-sales" className="w-full">
   229→        <TabsList className="grid grid-cols-4 w-full">
   230→          <TabsTrigger value="egg-sales"><ShoppingBasket className="h-3 w-3 mr-1" />Egg Sales</TabsTrigger>
   231→          <TabsTrigger value="bird-sales"><Bird className="h-3 w-3 mr-1" />Bird Sales</TabsTrigger>
   232→          <TabsTrigger value="customers"><Users className="h-3 w-3 mr-1" />Customers</TabsTrigger>
   233→          <TabsTrigger value="unpaid"><AlertCircle className="h-3 w-3 mr-1" />Unpaid</TabsTrigger>
   234→        </TabsList>
   235→
   236→        <TabsContent value="egg-sales">
   237→          <Card>
   238→            <CardHeader className="pb-2">
   239→              <CardTitle className="text-base">Record Egg Sale</CardTitle>
   240→            </CardHeader>
   241→            <CardContent>
   242→              <form onSubmit={submitEggSale} className="grid grid-cols-2 md:grid-cols-3 gap-3">
   243→                <div>
   244→                  <Label className="text-xs">Customer</Label>
   245→                  <Select value={eggSaleCustomerId} onValueChange={setEggSaleCustomerId}>
   246→                    <SelectTrigger className="h-9 text-sm"><SelectValue placeholder="Walk-in" /></SelectTrigger>
   247→                    <SelectContent>
   248→                      {customers.map(c => <SelectItem key={c.id} value={c.id}>{c.name}</SelectItem>)}
   249→                    </SelectContent>
   250→                  </Select>
   251→                </div>
   252→                <div>
   253→                  <Label className="text-xs">Farm *</Label>
   254→                  <Select value={eggSaleFarmId} onValueChange={setEggSaleFarmId}>
   255→                    <SelectTrigger className="h-9 text-sm"><SelectValue placeholder="Select farm" /></SelectTrigger>
   256→                    <SelectContent>
   257→                      {farms.map(f => <SelectItem key={f.id} value={f.id}>{f.name}</SelectItem>)}
   258→                    </SelectContent>
   259→                  </Select>
   260→                </div>
   261→                <div>
   262→                  <Label className="text-xs">Date</Label>
   263→                  <Input name="date" type="date" defaultValue={today} className="h-9 text-sm" />
   264→                </div>
   265→                <div>
   266→                  <Label className="text-xs">Crates *</Label>
   267→                  <Input name="crateCount" type="number" min="1" required placeholder="e.g. 10" className="h-9 text-sm" />
   268→                </div>
   269→                <div>
   270→                  <Label className="text-xs">Price per Crate (GHS) *</Label>
   271→                  <Input name="pricePerCrate" type="number" min="0" step="0.5" required placeholder="e.g. 35" className="h-9 text-sm" />
   272→                </div>
   273→                <div>
   274→                  <Label className="text-xs">Payment</Label>
   275→                  <Select value={eggSalePayment} onValueChange={setEggSalePayment}>
   276→                    <SelectTrigger className="h-9 text-sm"><SelectValue /></SelectTrigger>
   277→                    <SelectContent>
   278→                      <SelectItem value="Paid">Paid</SelectItem>
   279→                      <SelectItem value="Partial">Partial</SelectItem>
   280→                      <SelectItem value="Unpaid">Credit / Unpaid</SelectItem>
   281→                    </SelectContent>
   282→                  </Select>
   283→                </div>
   284→                <div className="flex items-end col-span-2 md:col-span-3">
   285→                  <Button type="submit" className="w-full md:w-auto h-9"><Plus className="h-3 w-3 mr-1" />Record Sale</Button>
   286→                </div>
   287→              </form>
   288→            </CardContent>
   289→          </Card>
   290→
   291→          {/* Recent egg sales */}
   292→          <Card className="mt-4">
   293→            <CardHeader className="pb-2">
   294→              <CardTitle className="text-base">Recent Egg Sales</CardTitle>
   295→            </CardHeader>
   296→            <CardContent>
   297→              <ScrollArea className="max-h-64">
   298→                <div className="space-y-2">
   299→                  {eggSales.slice(0, 20).map(s => (
   300→                    <div key={s.id} className="flex items-center gap-3 p-2 bg-gray-50 rounded-lg text-sm">
   301→                      <ShoppingBasket className="h-4 w-4 text-green-600 shrink-0" />
   302→                      <div className="flex-1">
   303→                        <p className="font-medium">{s.customer?.name || 'Walk-in'}</p>
   304→                        <p className="text-xs text-gray-500">{new Date(s.date).toLocaleDateString('en-GB')} · {s.crateCount} crates x {formatGHS(s.pricePerCrate)}</p>
   305→                      </div>
   306→                      <div className="text-right">
   307→                        <p className="font-medium">{formatGHS(s.totalAmount)}</p>
   308→                        <Badge variant={s.paymentStatus === 'Paid' ? 'default' : 'destructive'} className="text-[10px]">
   309→                          {s.paymentStatus}
   310→                        </Badge>
   311→                      </div>
   312→                      <Button size="sm" variant="ghost" className="h-7 w-7 p-0 shrink-0 text-gray-400 hover:text-amber-600" onClick={() => openAmendmentDialog('EggSale', s.id, [
   313→                        { key: 'crateCount', label: 'Crates', type: 'number' as const, value: s.crateCount },
   314→                        { key: 'pricePerCrate', label: 'Price/Crate', type: 'number' as const, value: s.pricePerCrate },
   315→                        { key: 'paymentStatus', label: 'Payment Status', type: 'text' as const, value: s.paymentStatus },
   316→                        { key: 'amountPaid', label: 'Amount Paid', type: 'number' as const, value: s.amountPaid },
   317→                      ])}>
   318→                        <Pencil className="h-3 w-3" />
   319→                      </Button>
   320→                    </div>
   321→                  ))}
   322→                </div>
   323→              </ScrollArea>
   324→            </CardContent>
   325→          </Card>
   326→        </TabsContent>
   327→
   328→        <TabsContent value="bird-sales">
   329→          <Card>

   332→            </CardHeader>
   333→            <CardContent>
   334→              <form onSubmit={submitBirdSale} className="grid grid-cols-2 md:grid-cols-3 gap-3">
   335→                <div>
   336→                  <Label className="text-xs">Customer</Label>
   337→                  <Select value={birdSaleCustomerId} onValueChange={setBirdSaleCustomerId}>
   338→                    <SelectTrigger className="h-9 text-sm"><SelectValue placeholder="Walk-in" /></SelectTrigger>
   339→                    <SelectContent>
   340→                      {customers.map(c => <SelectItem key={c.id} value={c.id}>{c.name}</SelectItem>)}

   343→                </div>

   348→                    <SelectContent>
   349→                      {farms.map(f => <SelectItem key={f.id} value={f.id}>{f.name}</SelectItem>)}
   350→                    </SelectContent>
   351→                  </Select>
   352→                </div>
   353→                <div>
   354→                  <Label className="text-xs">Date</Label>

   356→                </div>
   357→                <div>

   376→                      <SelectItem value="Unpaid">Credit / Unpaid</SelectItem>
   377→                    </SelectContent>
   378→                  </Select>
   379→                </div>
   380→                <div className="flex items-end col-span-2">
   381→                  <Button type="submit" className="w-full md:w-auto h-9"><Plus className="h-3 w-3 mr-1" />Record Sale</Button>

   386→
   387→          <Card className="mt-4">
   388→            <CardHeader className="pb-2">
   389→              <CardTitle className="text-base">Recent Bird Sales</CardTitle>
   390→            </CardHeader>
   391→            <CardContent>
   392→              <ScrollArea className="max-h-64">
   393→                <div className="space-y-2">
   394→                  {birdSales.slice(0, 20).map(s => (

   397→                      <div className="flex-1">

   400→                      </div>
   401→                      <div className="text-right">

   404→                          {s.paymentStatus}
   405→                        </Badge>
   406→                      </div>
   407→                      <Button size="sm" variant="ghost" className="h-7 w-7 p-0 shrink-0 text-gray-400 hover:text-amber-600" onClick={() => openAmendmentDialog('BirdSale', s.id, [
   408→                        { key: 'quantity', label: 'Quantity', type: 'number' as const, value: s.quantity },
