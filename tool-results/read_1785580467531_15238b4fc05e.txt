    31→// ============ FARM NAME EDITOR ============
    32→function FarmNameEditor() {
    33→  const { currentUser } = useAppStore()
    34→  const [farmName, setFarmName] = useState('')
    35→  const [savedName, setSavedName] = useState('')
    36→  const [editing, setEditing] = useState(false)
    37→  const [saving, setSaving] = useState(false)
    38→
    39→  const fetchSettings = useCallback(async () => {
    40→    try {
    41→      const res = await fetch('/api/reset', {
    42→        method: 'POST',
    43→        headers: { 'Content-Type': 'application/json' },
    44→        body: JSON.stringify({ action: 'getFarmName', role: currentUser?.role }),
    45→      })
    46→      if (res.ok) {
    47→        const data = await res.json()
    48→        setFarmName(data.farmName)
    49→        setSavedName(data.farmName)
    50→      }
    51→    } catch {}
    52→  }, [currentUser?.role])
    53→
    54→  useEffect(() => { fetchSettings() }, [fetchSettings])
    55→
    56→  const handleSave = async () => {
    57→    if (!farmName.trim() || farmName.trim().length < 2) {
    58→      toast.error('Farm name must be at least 2 characters')
    59→      return
    60→    }
    61→    setSaving(true)
    62→    try {
    63→      const res = await fetch('/api/reset', {
    64→        method: 'POST',
    65→        headers: { 'Content-Type': 'application/json' },
    66→        body: JSON.stringify({ action: 'updateFarmName', farmName: farmName.trim(), role: currentUser?.role }),
    67→      })
    68→      if (res.ok) {
    69→        toast.success('Farm name updated successfully')
    70→        setSavedName(farmName.trim())
    71→        setEditing(false)
    72→      } else {
    73→        const err = await res.json().catch(() => ({}))
    74→        toast.error(err.error || 'Failed to update farm name')
    75→      }
    76→    } catch {
    77→      toast.error('Network error')
    78→    } finally {
    79→      setSaving(false)
    80→    }
    81→  }
    82→
    83→  return (
    84→    <Card>
    85→      <CardHeader className="pb-2">
    86→        <CardTitle className="text-base flex items-center gap-2">
    87→          <Settings className="h-4 w-4 text-amber-600" />
    88→          Farm Name
    89→        </CardTitle>
    90→        <CardDescription className="text-xs">
    91→          This name appears across all dashboards, login screens, and reports.
    92→        </CardDescription>
    93→      </CardHeader>
    94→      <CardContent>
    95→        {editing ? (
    96→          <div className="space-y-3">
    97→            <div>
    98→              <Label className="text-xs">Farm / Business Name</Label>
    99→              <Input
   100→                value={farmName}
   101→                onChange={e => setFarmName(e.target.value)}
   102→                placeholder="Enter farm name"
   103→                className="h-9 text-sm"
   104→                autoFocus
   105→              />
   106→            </div>
   107→            <div className="flex gap-2">
   108→              <Button size="sm" onClick={handleSave} disabled={saving} className="h-8 text-xs">
   109→                {saving ? <Loader2 className="h-3 w-3 animate-spin" /> : <Check className="h-3 w-3" />}
   110→                Save
   111→              </Button>
   112→              <Button size="sm" variant="outline" onClick={() => { setFarmName(savedName); setEditing(false) }} className="h-8 text-xs">
   113→                <X className="h-3 w-3 mr-1" /> Cancel
   114→              </Button>
   115→            </div>
   116→          </div>
   117→        ) : (
   118→          <div className="flex items-center justify-between">
   119→            <div>
   120→              <p className="text-sm font-medium text-gray-900">{savedName}</p>
   121→              <p className="text-[10px] text-gray-500">Currently displayed across the system</p>
   122→            </div>
   123→            <Button size="sm" variant="outline" onClick={() => setEditing(true)} className="h-8 text-xs gap-1">
   124→              <Pencil className="h-3 w-3" /> Edit
   125→            </Button>
   126→          </div>
   127→        )}
   128→      </CardContent>
   129→    </Card>
   130→  )
   131→}

   417→// ============ BIRD FLOCK / POPULATION MANAGER ============
   418→function BirdFlockManager() {
   419→  const [farms, setFarms] = useState<FarmInfo[]>([])
   420→  const [flocks, setFlocks] = useState<any[]>([])
   421→  const [loading, setLoading] = useState(true)
   422→  const [saving, setSaving] = useState(false)
   423→  const [expandedFarm, setExpandedFarm] = useState<string | null>(null)
   424→
   425→  // Add flock form
   426→  const [addOpen, setAddOpen] = useState(false)
   427→  const [newFarmId, setNewFarmId] = useState('')
   428→  const [newName, setNewName] = useState('')
   429→  const [newType, setNewType] = useState('Layer')
   430→  const [newBreed, setNewBreed] = useState('Bovan Brown')
   431→  const [newCount, setNewCount] = useState('')
   432→  const [newAge, setNewAge] = useState('')
   433→  const [newDate, setNewDate] = useState(new Date().toISOString().split('T')[0])
   434→  const [newNotes, setNewNotes] = useState('')
   435→
   436→  const fetchData = useCallback(async () => {
   437→    setLoading(true)
   438→    try {
   439→      const [farmsRes, flocksRes] = await Promise.all([
   440→        fetch('/api/farms'),
   441→        fetch('/api/flocks'),
   442→      ])
   443→      if (farmsRes.ok) setFarms(await farmsRes.json())
   444→      if (flocksRes.ok) setFlocks(await flocksRes.json())
   445→    } catch {} finally {
   446→      setLoading(false)
   447→    }
   448→  }, [])
   449→
   450→  useEffect(() => { fetchData() }, [fetchData])
   451→
   452→  const handleAddFlock = async (e: React.FormEvent) => {
   453→    e.preventDefault()
   454→    if (!newFarmId || !newName.trim() || !newCount) {
   455→      toast.error('Farm, flock name, and bird count are required')
   456→      return
   457→    }
   458→    setSaving(true)
   459→    try {
   460→      const res = await fetch('/api/flocks', {
   461→        method: 'POST',
   462→        headers: { 'Content-Type': 'application/json' },
   463→        body: JSON.stringify({
   464→          farmId: newFarmId,
   465→          name: newName.trim(),
   466→          birdType: newType,
   467→          breed: newBreed,
   468→          birdCount: parseInt(newCount) || 0,
   469→          ageWeeks: parseInt(newAge) || 0,
   470→          dateArrival: newDate || new Date().toISOString(),
   471→          notes: newNotes.trim() || undefined,
   472→        }),
   473→      })
   474→      if (res.ok) {
   475→        toast.success(`Flock "${newName.trim()}" added with ${parseInt(newCount).toLocaleString()} birds`)
   476→        setAddOpen(false)
   477→        setNewName(''); setNewCount(''); setNewAge(''); setNewNotes('')
   478→        setNewDate(new Date().toISOString().split('T')[0])
   479→        fetchData()
   480→      } else {
   481→        const err = await res.json().catch(() => ({}))
   482→        toast.error(err.error || 'Failed to add flock')
   483→      }
   484→    } catch { toast.error('Network error') } finally { setSaving(false) }
   485→  }
   486→
   487→  const handleDeactivateFlock = async (flockId: string, flockName: string) => {
   488→    if (!confirm(`Deactivate flock "${flockName}"? The birds will no longer appear in counts, but all records are preserved.`)) return
   489→    setSaving(true)
   490→    try {
   491→      const res = await fetch('/api/flocks', {
   492→        method: 'PUT',
   493→        headers: { 'Content-Type': 'application/json' },
   494→        body: JSON.stringify({ id: flockId, isActive: false }),
   495→      })
   496→      if (res.ok) {
   497→        toast.success(`Flock "${flockName}" deactivated`)
   498→        fetchData()
   499→      } else {
   500→        toast.error('Failed to deactivate flock')
   501→      }
   502→    } catch { toast.error('Network error') } finally { setSaving(false) }
   503→  }
   504→
   505→  // Group flocks by farm
   506→  const farmsWithFlocks = farms.filter(f => f.isActive).map(farm => ({
   507→    ...farm,
   508→    flocks: flocks.filter(fl => fl.farmId === farm.id && fl.isActive),
   509→  }))
   510→  const totalBirds = flocks.filter(f => f.isActive).reduce((s, f) => s + f.birdCount, 0)
   511→  const totalInitial = flocks.filter(f => f.isActive).reduce((s, f) => s + (f.initialBirdCount || f.birdCount), 0)
   512→  const totalLosses = totalInitial - totalBirds
   513→
   514→  return (
   515→    <Card>
   516→      <CardHeader className="pb-2">
   517→        <div className="flex items-center justify-between">
   518→          <div>
   519→            <CardTitle className="text-base flex items-center gap-2">
   520→              <Bird className="h-4 w-4 text-purple-600" />
   521→              Bird Population per Farm

   730→function FullDataReset() {
   731→  const { currentUser } = useAppStore()
   732→  const [confirmStep, setConfirmStep] = useState(0)
   733→  const [saving, setSaving] = useState(false)
   734→
   735→  const handleFullReset = async () => {
   736→    setSaving(true)
   737→    try {
   738→      const res = await fetch('/api/reset', {
   739→        method: 'POST',
   740→        headers: { 'Content-Type': 'application/json' },
   741→        body: JSON.stringify({ role: currentUser?.role, action: 'fullReset', confirm: true }),
   742→      })
   743→      if (res.ok) {
   744→        const result = await res.json()
   745→        toast.success('All data has been wiped. The system is ready for a fresh start.')
   746→        setConfirmStep(0)
   747→        setTimeout(() => window.location.reload(), 1500)
   748→      } else {
   749→        const err = await res.json().catch(() => ({}))
   750→        toast.error(err.error || 'Failed to reset data')
   751→      }
   752→    } catch {
   753→      toast.error('Network error')
   754→    } finally {
   755→      setSaving(false)
   756→    }
   757→  }
   758→
   759→  return (
   760→    <Card className="border-red-200">
   761→      <CardHeader className="pb-2">
   762→        <CardTitle className="text-base flex items-center gap-2 text-red-700">
   763→          <RotateCcw className="h-4 w-4" />
   764→          Full Data Reset
   765→        </CardTitle>
   766→        <CardDescription className="text-xs">
   767→          Delete all records and start with a clean slate. This is irreversible.
   768→        </CardDescription>
   769→      </CardHeader>
   770→      <CardContent>
   771→        <div className="bg-red-50 border border-red-200 rounded-lg p-3 mb-3">
   772→          <p className="text-xs text-red-800">
   773→            <AlertTriangle className="h-3 w-3 inline mr-1" />
   774→            <strong>Warning:</strong> This will permanently delete ALL data from the system, including:
   775→          </p>
   776→          <ul className="text-[11px] text-red-700 mt-1 ml-4 list-disc">
   777→            <li>All egg collections, sales, and feed records</li>
   778→            <li>All bird flocks and mortality records</li>
   779→            <li>All expenses, treatments, and health checks</li>
   780→            <li>All vaccination records</li>
   781→            <li>All announcements and customers</li>
   782→            <li>All farm locations</li>
   783→            <li>All pending amendment requests</li>
   784→          </ul>
   785→          <p className="text-[11px] text-red-700 mt-1">
   786→            Your CEO account will remain but all other staff will be deactivated. You will need to re-add farms and staff.
   787→          </p>
   788→        </div>
   789→
   790→        {confirmStep === 0 ? (
   791→          <Button variant="destructive" size="sm" className="h-8 text-xs gap-1"
   792→            onClick={() => setConfirmStep(1)}>
   793→            <Trash2 className="h-3 w-3" /> Start Full Reset
   794→          </Button>
   795→        ) : (
   796→          <div className="space-y-3">
   797→            <div className="bg-yellow-50 border border-yellow-300 rounded-lg p-2.5">
   798→              <p className="text-xs font-medium text-yellow-900">
   799→                ⚠️ Are you absolutely sure?
   800→              </p>
   801→              <p className="text-[11px] text-yellow-800 mt-1">
   802→                This will permanently delete ALL data. Your CEO account will remain, but you will need to re-add farms and staff.
   803→              </p>
   804→            </div>
   805→            <div className="flex gap-2">
   806→              <Button variant="destructive" size="sm" onClick={handleFullReset} disabled={saving} className="h-8 text-xs gap-1">
   807→                {saving ? <Loader2 className="h-3 w-3 animate-spin" /> : <Trash2 className="h-3 w-3" />}
   808→                {saving ? 'Resetting...' : 'Yes, Delete Everything'}
   809→              </Button>
   810→              <Button variant="outline" size="sm" className="h-8 text-xs" onClick={() => setConfirmStep(0)}>
   811→                Cancel
   812→              </Button>
   813→            </div>
   814→          </div>
   815→        )}
   816→      </CardContent>
   817→    </Card>
