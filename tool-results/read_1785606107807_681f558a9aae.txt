    31→// ============ FARM NAME EDITOR ============
    32→function FarmNameEditor() {
    33→  const { currentUser } = useAppStore()
    34→  const [farmName, setFarmName] = useState('')
    35→  const [savedName, setSavedName] = useState('')
    36→  const [editing, setEditing] = useState(false)
    37→  const [saving, setSaving] = useState(false)
    38→
    39→  const fetchSettings = useCallback(async () => {
    40→    try {
    41→      const res = await fetch('/api/reset', {
    42→        method: 'POST',
    43→        headers: { 'Content-Type': 'application/json' },
    44→        body: JSON.stringify({ action: 'getFarmName', role: currentUser?.role }),
    45→      })
    46→      if (res.ok) {
    47→        const data = await res.json()
    48→        setFarmName(data.farmName)
    49→        setSavedName(data.farmName)
    50→      }
    51→    } catch {}
    52→  }, [currentUser?.role])
    53→
    54→  useEffect(() => { fetchSettings() }, [fetchSettings])
    55→
    56→  const handleSave = async () => {
    57→    if (!farmName.trim() || farmName.trim().length < 2) {
    58→      toast.error('Farm name must be at least 2 characters')
    59→      return
    60→    }
    61→    setSaving(true)
    62→    try {
    63→      const res = await fetch('/api/reset', {
    64→        method: 'POST',
    65→        headers: { 'Content-Type': 'application/json' },
    66→        body: JSON.stringify({ action: 'updateFarmName', farmName: farmName.trim(), role: currentUser?.role }),
    67→      })
    68→      if (res.ok) {
    69→        toast.success('Farm name updated successfully')
    70→        setSavedName(farmName.trim())
    71→        setEditing(false)
    72→      } else {
    73→        const err = await res.json().catch(() => ({}))
    74→        toast.error(err.error || 'Failed to update farm name')
    75→      }
    76→    } catch {
    77→      toast.error('Network error')
    78→    } finally {
    79→      setSaving(false)
    80→    }
    81→  }
    82→
    83→  return (
    84→    <Card>
    85→      <CardHeader className="pb-2">
    86→        <CardTitle className="text-base flex items-center gap-2">
    87→          <Settings className="h-4 w-4 text-amber-600" />
    88→          Farm Name
    89→        </CardTitle>
    90→        <CardDescription className="text-xs">
    91→          This name appears across all dashboards, login screens, and reports.
    92→        </CardDescription>
    93→      </CardHeader>
    94→      <CardContent>
    95→        {editing ? (
    96→          <div className="space-y-3">
    97→            <div>
    98→              <Label className="text-xs">Farm / Business Name</Label>
    99→              <Input
   100→                value={farmName}
   101→                onChange={e => setFarmName(e.target.value)}
   102→                placeholder="Enter farm name"
   103→                className="h-9 text-sm"
   104→                autoFocus
   105→              />
   106→            </div>
   107→            <div className="flex gap-2">
   108→              <Button size="sm" onClick={handleSave} disabled={saving} className="h-8 text-xs">
   109→                {saving ? <Loader2 className="h-3 w-3 animate-spin" /> : <Check className="h-3 w-3" />}
   110→                Save
   111→              </Button>
   112→              <Button size="sm" variant="outline" onClick={() => { setFarmName(savedName); setEditing(false) }} className="h-8 text-xs">
   113→                <X className="h-3 w-3 mr-1" /> Cancel
   114→              </Button>
   115→            </div>
   116→          </div>
   117→        ) : (
   118→          <div className="flex items-center justify-between">
   119→            <div>
   120→              <p className="text-sm font-medium text-gray-900">{savedName}</p>
   121→              <p className="text-[10px] text-gray-500">Currently displayed across the system</p>
   122→            </div>
   123→            <Button size="sm" variant="outline" onClick={() => setEditing(true)} className="h-8 text-xs gap-1">
   124→              <Pencil className="h-3 w-3" /> Edit
   125→            </Button>
   126→          </div>
   127→        )}
   128→      </CardContent>
   129→    </Card>
   130→  )
   131→}

   417→// ============ BIRD FLOCK / POPULATION MANAGER ============
   418→function BirdFlockManager() {
   419→  const [farms, setFarms] = useState<FarmInfo[]>([])
   420→  const [flocks, setFlocks] = useState<any[]>([])
   421→  const [farmLayRates, setFarmLayRates] = useState<Record<string, { todayLayRate: number; monthLayRate: number; todayEggs: number; monthAvgDailyEggs: number }>>({})
   422→  const [loading, setLoading] = useState(true)
   423→  const [saving, setSaving] = useState(false)
   424→  const [expandedFarm, setExpandedFarm] = useState<string | null>(null)
   425→  // Edit flock
   426→  const [editFlock, setEditFlock] = useState<any>(null)
   427→  const [editCount, setEditCount] = useState('')
   428→  const [editNotes, setEditNotes] = useState('')
   429→
   430→  // Add flock form
   431→  const [addOpen, setAddOpen] = useState(false)
   432→  const [newFarmId, setNewFarmId] = useState('')
   433→  const [newName, setNewName] = useState('')
   434→  const [newType, setNewType] = useState('Layer')
   435→  const [newBreed, setNewBreed] = useState('Bovan Brown')
   436→  const [newCount, setNewCount] = useState('')
   437→  const [newAge, setNewAge] = useState('')
   438→  const [newDate, setNewDate] = useState(new Date().toISOString().split('T')[0])
   439→  const [newNotes, setNewNotes] = useState('')
   440→
   441→  const fetchData = useCallback(async () => {
   442→    setLoading(true)
   443→    try {
   444→      const [farmsRes, flocksRes] = await Promise.all([
   445→        fetch('/api/farms'),
   446→        fetch('/api/flocks'),
   447→      ])
   448→      if (farmsRes.ok) setFarms(await farmsRes.json())
   449→      if (flocksRes.ok) {
   450→        const flocksData = await flocksRes.json()
   451→        // API now returns { flocks, farmLayRates }
   452→        if (Array.isArray(flocksData)) {
   453→          setFlocks(flocksData)
   454→        } else {
   455→          setFlocks(flocksData.flocks || [])
   456→          setFarmLayRates(flocksData.farmLayRates || {})
   457→        }
   458→      }
   459→    } catch {} finally {
   460→      setLoading(false)
   461→    }
   462→  }, [])
   463→
   464→  useEffect(() => { fetchData() }, [fetchData])
   465→
   466→  const handleAddFlock = async (e: React.FormEvent) => {
   467→    e.preventDefault()
   468→    if (!newFarmId || !newName.trim() || !newCount) {
   469→      toast.error('Farm, flock name, and bird count are required')
   470→      return
   471→    }
   472→    setSaving(true)
   473→    try {
   474→      const res = await fetch('/api/flocks', {
   475→        method: 'POST',
   476→        headers: { 'Content-Type': 'application/json' },
   477→        body: JSON.stringify({
   478→          farmId: newFarmId,
   479→          name: newName.trim(),
   480→          birdType: newType,
   481→          breed: newBreed,
   482→          birdCount: parseInt(newCount) || 0,
   483→          ageWeeks: parseInt(newAge) || 0,
   484→          dateArrival: newDate || new Date().toISOString(),
   485→          notes: newNotes.trim() || undefined,
   486→        }),
   487→      })
   488→      if (res.ok) {
   489→        toast.success(`Flock "${newName.trim()}" added with ${parseInt(newCount).toLocaleString()} birds`)
   490→        setAddOpen(false)
   491→        setNewName(''); setNewCount(''); setNewAge(''); setNewNotes('')
   492→        setNewDate(new Date().toISOString().split('T')[0])
   493→        fetchData()
   494→      } else {
   495→        const err = await res.json().catch(() => ({}))
   496→        toast.error(err.error || 'Failed to add flock')
   497→      }
   498→    } catch { toast.error('Network error') } finally { setSaving(false) }
   499→  }
   500→
   501→  const handleDeactivateFlock = async (flockId: string, flockName: string) => {
   502→    if (!confirm(`Deactivate flock "${flockName}"? The birds will no longer appear in counts, but all records are preserved.`)) return
   503→    setSaving(true)
   504→    try {
   505→      const res = await fetch('/api/flocks', {
   506→        method: 'PUT',
   507→        headers: { 'Content-Type': 'application/json' },
   508→        body: JSON.stringify({ id: flockId, isActive: false }),
   509→      })
   510→      if (res.ok) {
   511→        toast.success(`Flock "${flockName}" deactivated`)
   512→        fetchData()
   513→      } else {
   514→        toast.error('Failed to deactivate flock')
   515→      }
   516→    } catch { toast.error('Network error') } finally { setSaving(false) }
   517→  }
   518→
   519→  const handleEditFlock = (flock: any) => {
   520→    setEditFlock(flock)
   521→    setEditCount(String(flock.birdCount))
   522→    setEditNotes(flock.notes || '')
